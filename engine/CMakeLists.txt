cmake_minimum_required(VERSION 3.20)
project(remagx_core LANGUAGES C CXX)
if(EMSCRIPTEN)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/emscripten.version" required_emscripten)
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} --version OUTPUT_VARIABLE compiler_banner COMMAND_ERROR_IS_FATAL ANY)
  string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" actual_emscripten "${compiler_banner}")
  if(NOT actual_emscripten STREQUAL required_emscripten)
    message(FATAL_ERROR "Use pinned Emscripten ${required_emscripten}")
  endif()
  option(BUILD_SHARED_LIBS "Build a shared native library" OFF)
else()
  option(BUILD_SHARED_LIBS "Build a shared native library" ON)
endif()
add_library(remagx_core src/parratt.cpp src/angular_resolution.cpp src/table_lookup.cpp src/energy_scan.cpp src/dielectric_tensor.cpp)
target_include_directories(remagx_core PUBLIC include)
target_compile_features(remagx_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(remagx_core PRIVATE /W4 /fp:strict)
  set_target_properties(remagx_core PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
  target_compile_options(remagx_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
if(EMSCRIPTEN)
  if(BUILD_SHARED_LIBS)
    message(FATAL_ERROR "The WASM target requires BUILD_SHARED_LIBS=OFF")
  endif()
  target_compile_options(remagx_core PRIVATE -fexceptions)
  add_executable(remagx_wasm src/wasm_exports.cpp)
  target_link_libraries(remagx_wasm PRIVATE remagx_core)
  set_target_properties(remagx_wasm PROPERTIES OUTPUT_NAME "remagx-core" SUFFIX ".mjs")
  target_link_options(remagx_wasm PRIVATE --no-entry -fexceptions
    "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,worker,node"
    "-sEXPORTED_FUNCTIONS=['_remagx_parratt','_remagx_parratt_qz','_remagx_parratt_qz_resolution','_remagx_table_lookup','_remagx_energy_resolution','_remagx_parratt_energy_scan','_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['HEAPF64']"
    "-sALLOW_MEMORY_GROWTH=1" "-sABORTING_MALLOC=0")
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME double_precision_gate COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_precision.py)
  add_test(NAME precision_gate_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_precision_gate.py)
  add_executable(c_api_smoke tests/c_api_smoke.c)
  target_link_libraries(c_api_smoke PRIVATE remagx_core)
  add_test(NAME c_api_smoke COMMAND c_api_smoke)
  if(BUILD_SHARED_LIBS AND NOT EMSCRIPTEN)
    add_test(NAME dielectric_tensor_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_dielectric_tensor.py $<TARGET_FILE:remagx_core>)
    add_test(NAME energy_scan_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_energy_scan.py $<TARGET_FILE:remagx_core>)
    add_test(NAME energy_resolution_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_energy_resolution.py $<TARGET_FILE:remagx_core>)
    add_test(NAME tables_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tables.py $<TARGET_FILE:remagx_core>)
    add_test(NAME parratt_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parratt.py $<TARGET_FILE:remagx_core>)
  endif()
endif()

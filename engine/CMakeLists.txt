cmake_minimum_required(VERSION 3.20)
project(remagx_core LANGUAGES C CXX)
option(BUILD_SHARED_LIBS "Build a shared native library" ON)
add_library(remagx_core src/parratt.cpp)
target_include_directories(remagx_core PUBLIC include)
target_compile_features(remagx_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(remagx_core PRIVATE /W4 /fp:strict)
  set_target_properties(remagx_core PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
  target_compile_options(remagx_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME double_precision_gate COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_precision.py)
  add_test(NAME precision_gate_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_precision_gate.py)
  add_executable(c_api_smoke tests/c_api_smoke.c)
  target_link_libraries(c_api_smoke PRIVATE remagx_core)
  add_test(NAME c_api_smoke COMMAND c_api_smoke)
  if(BUILD_SHARED_LIBS AND NOT EMSCRIPTEN)
    add_test(NAME parratt_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parratt.py $<TARGET_FILE:remagx_core>)
  endif()
endif()
